%--- SLIDE 3: The Energy Trilemma ------------------------------------------
\begin{frame}{Vietnam's Energy Trilemma}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      Vietnam's growth model confronts three simultaneously binding constraints:
      \medskip
      \begin{itemize}
        \item \fuvhl{Energy security:} rapidly rising industrial demand
        \item \fuvhl{Sustainability:} Paris Agreement + Net Zero pledges
        \item \fuvhl{Affordability:} administratively regulated retail tariffs (EVN)
      \end{itemize}
      \medskip
      \begin{keybox}[title=PDP8 at a Glance]
        \small
        \begin{tabular}{@{}lll@{}}
          & \textbf{2024 actual} & \textbf{2030 target}\\
          \midrule
          Solar + Wind & 22.8 GW & 73.5 GW\\
          Battery BESS & 0.25 GW & 14.0 GW\\
          Grid invest.\ & \$3.2 B/yr & \$3.6 B/yr\\
        \end{tabular}
      \end{keybox}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{alertbox}[title=The Core Structural Friction]
        \small Renewable generation is \textit{non-dispatchable}: supply clears in real time with demand. As VRE penetration rises without adequate storage, the \textbf{reliability penalty}---endogenous reduction in capital utilization from grid instability---acts like a persistent negative TFP shock.
      \end{alertbox}
      \vspace{0.3em}
      \begin{goldbox}[title=Empirical Trigger]
        \small Solar capacity doubled in 18 months (2020–21) under FiT incentives, while battery storage remained near zero. Vietnam already experiences curtailment $>$10\% in southern provinces.
      \end{goldbox}
    \end{column}
  \end{columns}
\end{frame}

%--- SLIDE 4: Literature Gap -----------------------------------------------
\begin{frame}{Literature Gap: What Existing Models Miss}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{fuvblue}{What the literature covers:}}
      \begin{itemize}\small
        \item \textit{E-DSGE models} (Heutel 2012; Airaudo et al.\ 2025): carbon taxes, ``greenflation''
        \item \textit{Directed Technical Change} (Acemoglu et al.\ 2012): R\&D-race between sectors
        \item \textit{Techno-economic} (Zakeri \& Syri 2015; Schreiner 2020): storage costs, grid investment
        \item \textit{Open economy} (Schmitt-Grohé 2003): debt-elastic interest rates
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{\textcolor{fuvred}{Critical blind spots:}}
      \begin{itemize}\small
        \item No model \textit{endogenizes grid reliability} as a capital utilization constraint
        \item DTC literature ignores \textit{deployment-driven learning} in storage
        \item Public--private investment \textit{asymmetry} (``agility gap'') is not formalized
        \item Open-economy E-DSGE ignores \textit{battery import price} as a terms-of-trade shock
      \end{itemize}
      \vspace{0.3em}
      \begin{keybox}
        \small \textbf{This capstone} bridges all four gaps jointly---in a single DSGE calibrated to Vietnam.
      \end{keybox}
    \end{column}
  \end{columns}
\end{frame}

%--- SLIDE 7: Households ---------------------------------------------------
\begin{frame}{Households}
  A representative household maximizes discounted lifetime utility:
  \begin{equation}
    \max_{\{C_t,L_t,K_{p,t+1},B^*_t\}} \;\mathbb{E}_0 \sum_{t=0}^{\infty}\beta^t
    \!\left[\log C_t - \frac{L_t^{1+\sigma_L}}{1+\sigma_L}\right]
  \end{equation}
  subject to the \textbf{consolidated budget constraint}:
  \begin{equation}
    C_t + I_{p,t} + P_{bat,t}I_{bat,t} + B^*_t + T_t
    = Y_t + (1+r^*_{t-1})B^*_{t-1}
  \end{equation}
  \medskip
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{block}{Capital Euler Equation}
        \vspace{-0.3em}
        \[
          \frac{1}{C_t}=\beta\,\mathbb{E}_t\!\left[\frac{\alpha Y_{t+1}/K_{p,t+1}+1-\delta_p}{C_{t+1}}\right]
        \]
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Labor Supply}
        \vspace{-0.3em}
        \[
          (1-\alpha)V_t = \varphi\,C_t\,L_t^{1+\sigma_L}
        \]
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.4em}
  \begin{keybox}
    \small \textbf{Key modeling choice:} Capital utilization $u_t$ is \textit{not} a household decision variable. It reflects the fraction of installed capital that can be operated given grid reliability---an exogenous physical constraint, not a margin of adjustment.
  \end{keybox}
\end{frame}

%--- SLIDE 8: Production & Reliability ------------------------------------
\begin{frame}{Production Technology \& Endogenous Reliability}
  \textbf{Final output} (CES of value-added and energy):
  \[
    Y_t = \left[s_V\,V_t^{(\sigma_E-1)/\sigma_E} + (1-s_V)\bar{E}^{(\sigma_E-1)/\sigma_E}\right]^{\sigma_E/(\sigma_E-1)},
    \quad \sigma_E = 0.6
  \]
  \textbf{Value-added} (Cobb-Douglas with endogenous utilization):
  \[
    V_t = (u_t K_{p,t})^\alpha\,L_t^{1-\alpha}, \quad \alpha=0.35
  \]

  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{alertbox}[title=Reliability Constraint (Novel)]
        \[
          u_t = 1 - \exp\!\left(-\psi\,\frac{F_t}{\phi_{int}\cdot\mathrm{Vol}_{ren,t}}\right)
        \]
        \small When $F_t\uparrow$ (more flexibility) or $\mathrm{Vol}_{ren,t}\downarrow$, utilization rises toward 1. The exponential form captures the \textit{nonlinear} costs of reliability shortfalls.
      \end{alertbox}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{keybox}[title=CES Flexibility Bundle (Novel)]
        \[
          F_t = \!\left[\mu(A_{bat,t}K_{b,t})^{\frac{\rho_f-1}{\rho_f}}
          +(1{-}\mu)K_{g,t}^{\frac{\rho_f-1}{\rho_f}}\right]^{\frac{\rho_f}{\rho_f-1}}
        \]
        \small Batteries and grid are \textit{complements} ($\rho_f=0.4<1$): neither substitutes for the other when the opposing asset is scarce.
      \end{keybox}
    \end{column}
  \end{columns}
\end{frame}

%--- SLIDE 9: Battery Investment & Learning --------------------------------
\begin{frame}{Battery Investment Rule \& Learning-by-Doing}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{block}{Battery Investment (Private, Agile)}
        \[
          \frac{I_{bat,t}}{\bar{I}_{bat}} = \left(\frac{u^*}{u_t}\right)^{\phi_{grid}}\!\!\frac{1}{P_{bat,t}}
        \]
        \begin{itemize}\small
          \item Responds \textit{contemporaneously} to reliability gap
          \item Dampened by world battery price $P_{bat,t}$
          \item Capital accumulates: $K_{b,t+1}=(1-\delta_b)K_{b,t}+I_{bat,t}$
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{alertbox}[title=Endogenous Learning (Novel)]
        \[
          A_{bat,t} - A_{bat,t-1} = \eta_{bat}\cdot\chi\cdot\frac{u^*-u_t}{u^*}
        \]
        \begin{itemize}\small
          \item $\eta_{bat}=0.10$: learning speed (18--22\% Li-ion experience curve)
          \item $\chi\in[0,1]$: \fuvhl{regulatory transmission parameter}
          \item $\chi=1$: full price pass-through (liberalized market)
          \item $\chi=0$: fixed tariffs suppress scarcity signal
        \end{itemize}
      \end{alertbox}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \begin{goldbox}
    \small \textbf{The Agility Gap mechanism:} When $u_t < u^*$, battery investment responds \textit{immediately} (private) while grid investment requires multi-year construction lags (public). This differential speed is the source of the mid-transition reliability valley.
  \end{goldbox}
\end{frame}

%--- SLIDE 10: Grid Investment & External Sector ---------------------------
\begin{frame}{Public Grid Investment \& External Sector}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{block}{Grid Investment Rule (Government)}
        \[
          \hat{I}_{grid,t} = -\phi_{grid}\,\hat{u}_t + \varepsilon_{I,t}, \quad \phi_{grid}=1.5
        \]
        \begin{itemize}\small
          \item \textit{Reactive} policy: invests more when reliability falls
          \item Lump-sum tax $T_t$ finances grid; Ricardian equivalence holds
          \item $\varepsilon_{I,t}$: implementation delay shock (budget cycles, permitting)
          \item Capital: $K_{g,t+1}=(1-\delta_g)K_{g,t}+I_{grid,t}$
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Small Open Economy (Schmitt-Grohé 2003)}
        \textit{Current account / resource constraint:}
        \[
          B^*_t = (1+r^*_{t-1})B^*_{t-1} + Y_t - C_t - I_{p,t} - P_{bat,t}I_{bat,t} - I_{grid,t}
        \]
        \textit{Debt-elastic interest premium:}
        \[
          r^*_t = \bar{r} + \phi_b\!\left(e^{\bar{B}^*-B^*_t}-1\right)
        \]
        \begin{itemize}\small
          \item $\phi_b=0.001$ ensures stationarity of $B^*$
          \item Battery imports at world price $\Rightarrow$ terms-of-trade shock channel
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

%--- SLIDE 11: Shocks & Equilibrium ----------------------------------------
\begin{frame}{Exogenous Shocks \& Equilibrium Definition}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{block}{Three Exogenous Driving Forces}
        \begin{align}
          \hat{E}_{ren,t} &= \rho_{ren}\hat{E}_{ren,t-1}+\varepsilon^{ren}_t,\quad \varepsilon^{ren}\sim\mathcal{N}(0,\sigma_{ren}^2)\notag\\
          \hat{P}_{bat,t} &= \rho_{bat}\hat{P}_{bat,t-1}+\varepsilon^{bat}_t,\quad \varepsilon^{bat}\sim\mathcal{N}(0,\sigma_{bat}^2)\notag\\
          \varepsilon_{I,t} &= \rho_I\varepsilon_{I,t-1}+\nu_{I,t},\quad\nu_I\sim\mathcal{N}(0,\sigma_I^2)\notag
        \end{align}
        \begin{itemize}\small
          \item All shocks mutually independent
          \item $\rho_{ren}=0.85$ (monsoon seasonality)
          \item $\sigma_{ren}=0.12$ (CV of monthly VN solar output)
          \item $\rho_{bat}=0.90$, $\sigma_{bat}=0.08$ (lithium price cycle)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{keybox}[title={Competitive Equilibrium: 16 Equations, 16 Unknowns}]
        \scriptsize
        \begin{tabular}{@{}ll@{}}
          \toprule
          \textbf{Block} & \textbf{Equations}\\
          \midrule
          Households & Capital Euler, Bond Euler, Labor supply\\
          Production & CES output, Value-added\\
          Energy/Grid & Reliability $u_t$, Flexibility $F_t$\\
          Battery & $I_{bat}$ rule, $K_b$ accumulation, LBD $A_{bat}$\\
          Grid & $I_{grid}$ rule, $K_g$ accumulation\\
          Capital & $K_p$ accumulation\\
          External & Current account, Interest premium\\
          Shocks & Battery price AR(1)\\
          \bottomrule
        \end{tabular}
      \end{keybox}
      \vspace{0.2em}
      {\scriptsize \textbf{Blanchard-Kahn:} 3 eigenvalues outside the unit circle, matching 3 forward-looking variables ($C_t$, $Y_t$, $r^*_t$). Unique saddle-path equilibrium confirmed.}
    \end{column}
  \end{columns}
\end{frame}

%--- SLIDE 12: Calibration strategy ----------------------------------------
\begin{frame}{Calibration Strategy: Three-Stage Approach}
  \begin{enumerate}
    \item \textbf{Standard structural parameters} matched to long-run Vietnamese macroeconomic aggregates (GSO 2024, ADB, World Bank)
    \item \textbf{Reliability \& intermittency parameters} micro-founded on engineering data: NLDC facility-level generation, IEA Vietnam Energy Outlook 2023, EVN annual reliability reports
    \item \textbf{Learning-by-doing parameters} anchored to empirical experience curve literature for Li-ion batteries (Kittner et al.\ 2017; Way et al.\ 2022)
  \end{enumerate}
  \vspace{0.5em}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{keybox}[title={Calibration Target: Flexibility Weight $\mu=0.16$}]
        \scriptsize
        From PDP8 investment allocations:\medskip\\
        $K_{b,ss} = \frac{2.4/10}{0.12}\approx 2.0$ B USD (BESS)\\[2pt]
        $K_{g,ss} = \frac{14.9/10}{0.05}\approx 29.8$ B USD (grid)\\[4pt]
        With $\rho_f=0.4$: $\mu\approx 0.16$. Reflects the \textit{small but growing} share of private storage in Vietnam's current infrastructure.
      \end{keybox}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{keybox}[title={Complementarity: $\rho_f=0.40$}]
        \scriptsize Transmission moves energy across \textit{space}; storage moves it across \textit{time}. When solar output is zero (night), infinite transmission capacity is useless. Hart (2019) and Hirth (2013) document strong complementarity empirically---$\rho_f=0.4$ enforces this binding constraint.
      \end{keybox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Steady State Characterization}

        \end{enumerate}

%--- SLIDE 17: Log-linearization block 1 ----------------------------------
\begin{frame}{Log-Linearized System: Household \& Production Block}
  Variables expressed as log-deviations: $\hat{x}_t \equiv \ln X_t - \ln X_{ss}$
  \medskip
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{block}{Household Block}
        \textit{Capital Euler:}
        \[\hh{C}_t = \mathbb{E}_t\hh{C}_{t+1}
           -\frac{\alpha r_{p,ss}}{1+r_{p,ss}}\mathbb{E}_t\hh{Y}_{t+1}
           +\frac{\alpha r_{p,ss}}{1+r_{p,ss}}\mathbb{E}_t\hh{K}_{p,t+1}\]
        \textit{Labor supply:}
        \[\hh{V}_t = \hh{C}_t + (1+\sigma_L)\hh{L}_t\]
        \textit{Bond Euler:}
        \[\hh{C}_t = \mathbb{E}_t\hh{C}_{t+1}
           -\frac{r^*_{ss}}{1+r^*_{ss}}\mathbb{E}_t\hh{r}^*_{t+1}\]
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{Production \& Reliability Block}
        \textit{CES output:}
        \[\hh{Y}_t = s_V^{\sigma_E}\hh{V}_t
          + (1-s_V^{\sigma_E})\cdot\underbrace{0}_{\bar{E}\text{ fixed}}\]
        \textit{Value-added:}
        \[\hh{V}_t = \alpha(\hh{u}_t+\hh{K}_{p,t})+(1-\alpha)\hh{L}_t\]
        \textit{Reliability (linearized around $u^*$):}
        \[\hh{u}_t = -\xi\,\hh{F}_t + \xi\,\hh{\mathrm{Vol}}_{ren,t}\]
        where $\xi\equiv\psi(1-u_{ss})/(\phi_{int}\cdot\mathrm{Vol}_{ren,ss})$
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

%--- SLIDE 18: Log-linearization block 2 ----------------------------------
\begin{frame}{Log-Linearized System: Energy, Innovation \& External Block}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{block}{Flexibility \& Investment Block}
        \textit{CES flexibility (linearized):}
        \[\hh{F}_t = s_b(\hh{A}_{bat,t}+\hh{K}_{b,t}) + (1-s_b)\hh{K}_{g,t}\]
        \textit{Battery investment rule:}
        \[\hh{I}_{bat,t} = -\phi_{grid}\,\hh{u}_t - \hh{P}_{bat,t}\]
        \textit{Grid investment rule:}
        \[\hh{I}_{grid,t} = -\phi_{grid}\,\hh{u}_t + \varepsilon_{I,t}\]
        \textit{Innovation equation:}
        \[\hh{A}_{bat,t}-\hh{A}_{bat,t-1} = -\eta_{bat}\cdot\chi\cdot\hh{u}_t\]
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{block}{External Sector Block}
        \textit{Current account:}
        \begin{align*}
          B^*_t &= (1+r^*_{ss})B^*_{t-1} + Y_{ss}\hh{Y}_t - C_{ss}\hh{C}_t\\
          &\quad - I_{p,ss}\hh{I}_{p,t} - P_{bat,ss}I_{bat,ss}(\hh{P}_{bat,t}+\hh{I}_{bat,t})\\
          &\quad - I_{grid,ss}\hh{I}_{grid,t}
        \end{align*}
        \textit{Interest premium:}
        \[\hh{r}^*_t = -\frac{\phi_b}{r^*_{ss}}B^*_t\]
      \end{block}
      \begin{alertbox}
        \small ``Twin drain'': an intermittency shock simultaneously reduces $Y_t$ and raises battery import demand $I_{bat,t}$, creating a double hit on $B^*$.
      \end{alertbox}
    \end{column}
  \end{columns}
\end{frame}

%--- SLIDE 19: Transmission channels --------------------------------------
\begin{frame}{Three Transmission Channels}
  \begin{block}{Channel 1: The Reliability Wedge (Output Loss)}
    $\varepsilon^{ren}_t\uparrow \;\Rightarrow\; \mathrm{Vol}_{ren,t}\uparrow \;\Rightarrow\; u_t\downarrow \;\Rightarrow\; V_t\downarrow \;\Rightarrow\; Y_t\downarrow$

    {\small Impact elasticity: $\partial\hat{Y}/\partial\varepsilon^{ren}\approx -\alpha\xi s_V^{(\sigma_E-1)/\sigma_E}(1-\omega_E) < 0$. Operates like a negative TFP shock: supply-side, persistent, amplified by CES complementarity $\sigma_E<1$.}
  \end{block}
  \vspace{0.2em}
  \begin{block}{Channel 2: The Agility Gap (Asymmetric Investment)}
    $u_t < u^* \;\Rightarrow\; I_{bat,t}\uparrow$ (\textit{agile}); $I_{grid,t}\uparrow$ but with 5--7 year construction lags (\textit{sluggish})

    {\small Battery investment responds contemporaneously; grid capital accumulates with a bureaucratic time-to-build lag captured in the capital accumulation equation.}
  \end{block}
  \vspace{0.2em}
  \begin{block}{Channel 3: Reliability-Gap Learning (Innovation)}
    $u_t\downarrow \;\Rightarrow\; (u^*-u_t)/u^*\uparrow \;\Rightarrow\; \Delta A_{bat}\uparrow$ (if $\chi>0$)

    {\small Reliability scarcity triggers endogenous cost reduction in storage. Conditional on $\chi$: when regulated tariffs suppress scarcity signals, learning stalls. The ``valley of death'' for battery technology arises endogenously.}
  \end{block}
\end{frame}

%--- SLIDE 20: Solution method -------------------------------------------
\begin{frame}{Solution Method: First-Order Perturbation (Dynare~6)}
  The model is solved using \textbf{first-order perturbation} around the deterministic steady state, implemented in \textit{Dynare 6} (Adjemian et al.\ 2024):

  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{block}{State-Space Representation}
        \begin{align*}
          \mathbf{s}_{t+1} &= \mathbf{M}\,\mathbf{s}_t + \mathbf{N}\,\boldsymbol{\varepsilon}_t\\
          \mathbf{y}_t     &= \mathbf{P}\,\mathbf{s}_t
        \end{align*}
        \textbf{State vector:}
        \[\mathbf{s}_t = (\hh{K}_{p,t}, \hh{K}_{b,t}, \hh{K}_{g,t}, \hh{A}_{bat,t}, \hh{P}_{bat,t}, B^*_{t-1}, \hh{\varepsilon}_{ren,t})^\top\]
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{keybox}[title=Blanchard-Kahn Conditions]
        \small \textbf{3 eigenvalues outside the unit circle:}\\
        $\lambda_1=1.033$, $\lambda_2=-2.229$, $\lambda_3=\infty$\\[4pt]
        Matches \textbf{3 forward-looking variables}: $C_t$, $Y_t$, $r^*_t$\\[4pt]
        \textbf{7 stable eigenvalues:}\\
        $0,\;\approx0,\;0.90,\;0.97,\;0.98,\;0.99,\;0.99$\\[4pt]
        $\Rightarrow$ Unique saddle-path stable solution \checkmark
      \end{keybox}
    \end{column}
  \end{columns}
  \vspace{0.5em}
  \begin{alertbox}
    \small \textbf{Why first-order suffices:} The exponential nonlinearity of $u_t$ has been absorbed into the linearization coefficient $\xi$. The \textit{Reliability Valley} (transition path) is analyzed via a separate deterministic simulation to capture nonlinear effects beyond the linearized neighborhood.
  \end{alertbox}
\end{frame}

\begin{frame}{Summary: Three Novel Contributions to the Literature}
  \begin{enumerate}
    \item \textbf{Endogenous Grid Reliability as a Macro Friction}\\[2pt]
    {\small Capital utilization $u_t = 1-\exp(-\psi F_t/\phi_{int}\mathrm{Vol}_{ren,t})$ is determined endogenously by the stock of flexibility assets relative to VRE volatility. This transforms an engineering concept into a macroeconomic wedge in the aggregate production function---a channel absent from all extant E-DSGE models.}
    \medskip

    \item \textbf{Reliability-Gap-Induced Learning-by-Doing}\\[2pt]
    {\small Battery technology evolves endogenously: $\Delta A_{bat} = \eta_{bat}\cdot\chi\cdot(u^*-u)/u^*$. Unlike DTC models that require explicit R\&D sectors, innovation here responds to \textit{deployment-driven scarcity}. The regulatory parameter $\chi$ maps directly to observable policy instruments (tariff pass-through, capacity market design).}
    \medskip

    \item \textbf{Identification of the Structural Agility Gap}\\[2pt]
    {\small Private batteries (agile, price-driven) and public grid capital (sluggish, rule-based) are formally distinguished as \textit{complementary} inputs in a CES flexibility bundle ($\rho_f=0.4$). This generates a predictable mid-transition reliability valley---a quantitative prediction unavailable in models that treat flexibility as homogeneous.}
  \end{enumerate}
\end{frame}

        \item Battery invest: $\hat{I}_{bat,t} = -\phi_{grid}\hat{u}_t - \hat{P}_{bat,t}$

        \item Battery capital: $\hat{K}_{b,t+1}=(1-\delta_b)\hat{K}_{b,t}+\frac{I_{bat,ss}}{K_{b,ss}}\hat{I}_{bat,t}$
        \item Learning: $\hat{A}_{bat,t}-\hat{A}_{bat,t-1}=-\eta_{bat}\cdot\chi\cdot\hat{u}_t$
        \item Grid invest: $\hat{I}_{grid,t}=-\phi_{grid}\hat{u}_t+\varepsilon_{I,t}$
        \item Grid capital: $\hat{K}_{g,t+1}=(1-\delta_g)\hat{K}_{g,t}+\frac{I_{grid,ss}}{K_{g,ss}}\hat{I}_{grid,t}$
        \item Prod.\ capital: $\hat{K}_{p,t+1}=(1-\delta_p)\hat{K}_{p,t}+\frac{I_{p,ss}}{K_{p,ss}}\hat{I}_{p,t}$
        \item Current account: [linearized $B^*_t$ equation]
        \item Interest premium: $\hat{r}^*_t = -\frac{\phi_b}{r^*_{ss}}B^*_t$
        \item Battery price AR(1): $\hat{P}_{bat,t}=\rho_{bat}\hat{P}_{bat,t-1}+\varepsilon^{bat}_t$
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Appendix: Welfare Cost Formula}
  \begin{block}{Consumption-Equivalent Variation $\Lambda$ (Lucas 1987)}
    The welfare cost of intermittency is defined as the permanent percentage reduction in steady-state consumption that leaves the household indifferent between living in the stochastic economy vs.\ the deterministic steady state:
    \[
      \Lambda = 1 - \exp\!\left[\frac{1-\beta}{1-\beta^{T+1}}\sum_{t=0}^{T}\beta^t
      \log\!\left(\frac{C_t^{shock}}{C_{ss}}\right)\right]
    \]
    where $T=40$ quarters (10 years) and $C_t^{shock}$ follows the IRF path.
  \end{block}
  \vspace{0.5em}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{block}{Proposition: $\Lambda(\chi)$ Strictly Decreasing}
        Higher signal transmission $\chi$ $\Rightarrow$ faster $A_{bat}$ improvement $\Rightarrow$ higher $F_t$ $\Rightarrow$ higher $u_t$ $\Rightarrow$ higher $Y_t$ $\Rightarrow$ higher $C_t$ $\Rightarrow$ lower $\Lambda$.\\[4pt]
        {\small Proof: by monotone comparative statics on the linearized innovation equation.}
      \end{block}
    \end{column}
